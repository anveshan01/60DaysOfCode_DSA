class Pair{
    int r;
    int c;
    Pair(int i,int j){
        r=i;
        c=j;
    }
}
class Solution {
    public void solve(char[][] board) {
        int n=board.length;
        int m=board[0].length;
        Queue<Pair>q=new LinkedList<>();
        int[][]visited=new int[n][m];

        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(i==0 || j==0 || i==n-1 || j==m-1){
                    if(board[i][j]=='O'){
                    visited[i][j]=1;
                    q.offer(new Pair(i,j));
                    }
                }
            }
        }
        while(!q.isEmpty()){
            int row=q.peek().r;
            int col=q.peek().c;
            q.remove();
            int []drow={-1,0,1,0};
            int[] dcol={0,1,0,-1};
            for(int i=0;i<4;i++){
                int nr=row+drow[i];
                int nc=col+dcol[i];
                if(nr>=0 && nc>=0 && nc<m && nr<n && board[nr][nc]=='O' && visited[nr][nc]!=1){
                    q.offer(new Pair(nr,nc));
                    visited[nr][nc]=1;
                }
            }
        }
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(visited[i][j]!=1 && board[i][j]=='O'){
                    board[i][j]='X';
                }
            }
        }
    }
}
