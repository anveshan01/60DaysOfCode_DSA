class Solution {
    public int findKthLargest(int[] nums, int k) {
        PriorityQueue<Integer>pq=new PriorityQueue<>(Collections.reverseOrder());
        for(int i=0;i<nums.length;i++){
            pq.add(nums[i]);
        }
        while(!pq.isEmpty()){
            if(k==1)return pq.poll();
            else{
                pq.poll();
                k--;
            }

        }
        return -1;
    }
}
