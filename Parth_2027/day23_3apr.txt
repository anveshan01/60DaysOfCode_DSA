class Solution {
    public int removeElement(int[] nums, int val) {
        int n=nums.length;
        int[]ans=new int[n];
        int i=0;
        int k=0;
        int count=0;
        while(i<n){
            if(nums[i]==val){i++;}
            else{
                ans[k]=nums[i];
                k++;
                i++;
                count++;
            }
        }
        for(int j=0;j<n;j++){
            nums[j]=ans[j];
        }
        return count;
    }
}
