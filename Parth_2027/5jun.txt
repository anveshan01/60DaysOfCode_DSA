class Solution {
    public int findCenter(int[][] edges) {
        int n=edges.length;
        int a=edges[0][0];
        int b=edges[0][1];
        int c=edges[1][0];
        int d=edges[1][1];
        if(a==c){
            return a;
        }else if(a==d){
            return a;
        }else if(b==c){
            return b;
        }else if(b==d){
            return b;
        }else{
            return -1;
        }

    }
}
