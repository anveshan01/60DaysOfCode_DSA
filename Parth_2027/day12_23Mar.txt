//BRUTE FORCE SOL
class Solution {
    public int[] nextGreaterElement(int[] nums1, int[] nums2) {
        int [] ans=new int[nums1.length];
        int pt=0;
        for(int i=0;i<nums1.length;i++){//1
            int a=nums1[i];//1
            for(int j=0;j<nums2.length;j++){//0
                int b=nums2[j];//1
                if(a==b){//true
                boolean found=false;
                    for(int k=j+1;k<nums2.length;k++){//k=1
                        if(nums2[k]>b){//3>1
                            ans[pt]=nums2[k];//ans[1]=nums2[1]->3
                            pt++;//2
                            found=true;
                            break;
                        }
                    }
                    if(!found){
                    ans[pt]=-1;
                    pt++;//1
                    break;
                    }
                }
            }

        }
        return ans;
    }
}
