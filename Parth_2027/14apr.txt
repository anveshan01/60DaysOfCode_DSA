class Solution {
    public int candy(int[] ratings) {
        int n=ratings.length;
        int []ccount=new int[n];
        //for(int i=0;i<n;i++) ccount[i]=1; //all have 1 candy to start with
        Arrays.fill(ccount, 1);
        //left traverse
        for(int i=1;i<n;i++){
            if(ratings[i]>ratings[i-1]){
                ccount[i]=ccount[i-1]+1;
            }
        }
        //right traverse
        for(int i=n-2;i>=0;i--){
            if(ratings[i]>ratings[i+1]){
                ccount[i]=Math.max(ccount[i],ccount[i+1]+1);
            }
        }
        //count total candies
        int count=0;
        for(int i=0;i<n;i++){
            count+=ccount[i];
        }
        return count;

    }
}
