class Solution {
    public int climbStairs(int n) {
        if(n==0 || n==1){
            return 1;
        }
        int first=1;
        int second=2;
        if(n==2)return second;
        int ans=3;

        while(true){
            int sum=first+second;
            if(ans==n)return sum;
            first=second;
            second=sum;
            ans++;
        }

    }
}
