class Solution {
    public int mySqrt(int x) {
        if(x==1)return 1;
        if(x==0)return 0;
        int low=0;
        int high=x;
        while(low<=high){
            int mid=(low+high)/2;
            long ans=mid*mid;
            if(x/mid==mid){
                return mid;
            }else if(x/mid<mid){
                high=mid-1;
            }else{
                low=mid+1;
            }
        }
        return high;
    }
}
