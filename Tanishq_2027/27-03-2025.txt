    public int minimumIndex(List<Integer> nums) {
        int size = nums.size();
        HashMap<Integer, Integer> left = new HashMap<>();
        HashMap<Integer, Integer> right = new HashMap<>();
        for(int i = 0; i<size; i++){ // populate right map
            int curr = nums.get(i);
            if(right.get(curr) == null){ // new ele
                right.put(curr,1);
            }else{
                right.put(curr,(right.get(curr)+1));
            }
        }
        for(int i = 0; i<size; i++){ // dec from right and inc in left
            int curr = nums.get(i);
            if(left.get(curr) == null){ // new ele
                left.put(curr,1);
            }else{
                left.put(curr,(left.get(curr)+1));
            }
            right.put(curr,(right.get(curr)-1));

            if(left.get(curr) > (i+1)/2 && right.get(curr) > (size-i-1)/2){
                return i;
            }
        }
        return -1;
    }
}
