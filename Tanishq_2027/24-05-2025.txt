class Solution {
    public List<Integer> findWordsContaining(String[] words, char x) {
        List<Integer> list = new ArrayList<>();
        int idx = 0; 
        for(String i : words){
            
            
                if(i.contains(String.valueOf(x)) && !list.contains(idx)){
                    list.add(idx);
                }
            
            idx++;
        }
        return list;
    }
}
