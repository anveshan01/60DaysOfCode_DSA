class Solution {
    public int[] nextGreaterElement(int[] nums1, int[] nums2) {
        int len1 = nums1.length;
        int len2 = nums2.length;
        int[] res = new int[len1];
        for(int i = 0; i<len1; i++){
            int curr = nums1[i];
            int j = 0;
            while(j<len2 && nums1[i] != nums2[j]){
                j++;
            }
            while(j<len2){
                if(nums2[j]>curr){
                    res[i] = nums2[j];
                    break;
                }
                j++;
            }
            if(j == len2){res[i] = -1;}
        }
        return res;
    }
}
