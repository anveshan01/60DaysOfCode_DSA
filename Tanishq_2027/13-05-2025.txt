class Solution {
    public int lengthAfterTransformations(String s, int t) {
        int mod = 1000000007;
        int[] arr = new int[26];
        for(char curr : s.toCharArray()){
            arr[curr-'a']++;
        }
        for(int j = 0; j<t; j++){
            int[] next = new int[26];
            next[0] = arr[25];
            next[1] = (arr[25]+arr[0]) % mod;
            for(int i = 2; i<26; i++){
                next[i] = arr[i-1];
            }
            arr = next;
        }
        int ans = 0;
        for(int i = 0; i<26; i++){
            ans = (ans + arr[i]) % mod;
        }
        return ans;
    }
}
