class Solution {
    public void nextPermutation(int[] nums) {
        int i = nums.length-2;
        while(i>=0 && nums[i+1]<=nums[i]){
            i--;
        }
        if(i>=0){
            int j = nums.length-1;
            while(nums[j] <= nums[i]){
                j--;
            }
            int temp = nums[i]; //swap
            nums[i] = nums[j];
            nums[j] = temp;
        }
        //rev
        int k = nums.length-1;
        i = i+1;
        while(i < k){
            int temp = nums[i]; 
            nums[i] = nums[k];
            nums[k] = temp;
            i++; k--;
        }
    }
}
