class Solution {
    public boolean validMountainArray(int[] arr) {
        if(arr.length<3){return false;}
        int len = arr.length;
        int i = 0;
        while(i < len-1 && arr[i]<arr[i+1]){ //inc to mt
            i++;
        }
        if(i == len-1 || i == 0){
            return false;
        }
        while(i < len-1 && arr[i]>arr[i+1]){ //dec from mt
            i++;
        }
        if(i == len-1){
            return true;
        }
        return false;
    }
}
