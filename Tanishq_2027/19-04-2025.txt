class Solution {
    public long findPairs(int[] nums, int val){
        long res = 0;
        int left = 0, right = nums.length-1;
        while(left<right){
            if(nums[left]+nums[right] < val){
                res += (right-left);
                left++;
            }else{
                right--;
            }
        }
        return res;
    }
    public long countFairPairs(int[] nums, int lower, int upper) {
        Arrays.sort(nums);
        long low = findPairs(nums, lower);
        long high = findPairs(nums, upper+1);
        return high-low;
    }
}
