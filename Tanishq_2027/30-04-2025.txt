    public String tree2str(TreeNode root) {
        String res = String.valueOf(root.val);
        TreeNode left = root.left;
        TreeNode right = root.right;
        if(left != null){
            res += "(" + tree2str(left) + ")";
        } 
        if(right != null){
            if(root.left == null){
                res += "()";
            }
            res += "(" + tree2str(right) + ")";
        }
        return res;
    }
