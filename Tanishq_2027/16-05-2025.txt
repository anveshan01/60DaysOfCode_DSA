class Solution {
    public int minimumLength(String s) {
        int len = s.length();
        if (len == 1) {
            return 1;
        }
        int left = 0, right = len-1;
        while(left<right){
            if(s.charAt(left) != s.charAt(right)){
                return right-left+1;
            }
            else{ //if both equal
                char curr = s.charAt(left);
                while(left<len && s.charAt(left) == curr){left++;}
                while(right>0 && s.charAt(right) == curr){right--;}
            }
        }
        return ((right-left+1)>0)? (right-left+1) : 0;
    }
}

class Solution {
    public int mod = 1000000007;
    public int substr(int len){
        int res = 0;
        int i = 0;
        while(len-i > 1){
            res = (res+len-i) % mod;
            i++;
        }
        return (res+1) % mod;
    }
    public int countHomogenous(String s) {
        int length = s.length();
        int i = 0;
        int res = 0;
        while(i<length){
            int len = 1;
            while(i+1<length && s.charAt(i+1) == s.charAt(i)){
                len++; i++;
            }
            res = (res + substr(len)) % mod;
            i++;
        }
        return res;
    }
}
