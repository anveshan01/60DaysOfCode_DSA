class Solution {
    public List<String> printVertically(String s) {
        List<String> list = new ArrayList<>();
        String[] arr = s.split(" ");
        int max = 0;
        for(String currStr : arr){
            max = Math.max(max, currStr.length());
        }
        char[][] str = new char[arr.length][max];
        for(int i = 0; i<arr.length; i++){
            String curr = arr[i];
            for(int j = 0; j<curr.length(); j++){
                str[i][j] = curr.charAt(j);
            }
        }
        // print - WORKS
        // for(int i = 0; i<arr.length; i++){
        //     String curr = arr[i];
        //     for(int j = 0; j<curr.length(); j++){
        //         System.out.print(str[i][j]);
        //     }
        //     System.out.println();
        // }
        char[][] newStr = new char[max][arr.length];
        //System.out.println(max + " " + arr.length);
        for(int i = 0; i<max; i++){
            //String curr = arr[i];
            for(int j = 0; j<arr.length; j++){
                newStr[i][j] = str[j][i];
            }
        }
        //print
        // for(int i = 0; i<max; i++){
        //     //String curr = arr[i];
        //     for(int j = 0; j<arr.length; j++){
        //         System.out.print(newStr[i][j]);
        //     }
        //     System.out.println();
        // }
        for(int i = 0; i<newStr.length; i++){
            StringBuilder curr = new StringBuilder();
            //String curr = "";
            for(int j = 0; j<newStr[i].length; j++){
                if(newStr[i][j] == '\u0000'){
                    curr.append(" ");
                }
                else{
                    curr.append(newStr[i][j]);
                }
            }
            String currStr = curr.toString();
            currStr = currStr.replaceAll("\\s+$", ""); 
            //list.add(curr.toString().trim());
            list.add(currStr);
        }
        return list;
    }
}

class Solution {
    public void setZeroes(int[][] matrix) {
        int row = matrix.length;
        int col = matrix[0].length;
        HashSet<Integer> rows = new HashSet<>();
        HashSet<Integer> cols = new HashSet<>();
        for(int i = 0; i<row; i++){
            for(int j = 0; j<col; j++){
                if(matrix[i][j] == 0){
                    rows.add(i); cols.add(j);
                }
            }
        }
        for(int i = 0; i<row; i++){
            for(int j = 0; j<col; j++){
                if(rows.contains(i) || cols.contains(j)){
                    matrix[i][j] = 0;
                }
            }
        }
    }
}
